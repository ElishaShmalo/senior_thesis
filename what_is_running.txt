for L_val in np.array(num_unit_cells_vals) * N_val:
    fig, axs = plt.subplots(
        1, len(a_vals),
        figsize=(4 * len(a_vals), 4)
    )

    # Handles the case where there is only one a_val
    axs = np.atleast_1d(axs)

    for ax, a_val in zip(axs, a_vals):

        t_stars = np.array(
            find_t_star_lyp(L_val, a_val, [], show=False, skip_init=3, threshold=1/10)
        )

        # Poisson fit
        # lam = np.mean(t_stars)
        lam = poisson_fit(t_stars)

        xs = np.arange(0, t_stars.max() + 1)
        pmf = poisson.pmf(xs, lam)

        bins = np.arange(-0.5, t_stars.max() + 1.5, 1)

        ax.hist(
            t_stars,
            bins=bins,
            alpha=0.3,
            density=True
        )

        ax.plot(
            xs,
            pmf,
            "--",
            label=fr"$\mu={lam:.3f}$"
        )
        ax.set_title(rf"a = {a_val}")

    # plt.legend()
    axs[0].set_ylabel("Probability")
    fig.suptitle(fr"$L={L_val}$")

    plt.tight_layout()
    plt.show()
